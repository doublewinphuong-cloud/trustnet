--------------------------------------------------- -- 1. 建表部分 --------------------------------------------------- -- 角色表 CREATE TABLE Roles ( RoleID INT PRIMARY KEY AUTO_INCREMENT, RoleName VARCHAR(50) NOT NULL, Permissions TEXT ); -- 用户表 CREATE TABLE Users ( UserID INT PRIMARY KEY AUTO_INCREMENT, Name VARCHAR(100) NOT NULL, Email VARCHAR(150) UNIQUE NOT NULL, Age INT, Department VARCHAR(100), RoleID INT, FOREIGN KEY (RoleID) REFERENCES Roles(RoleID) ); -- 政策规则表 CREATE TABLE Policies ( PolicyID INT PRIMARY KEY AUTO_INCREMENT, Title VARCHAR(200) NOT NULL, Description TEXT, EffectiveDate DATE, Status VARCHAR(20) DEFAULT 'Active' ); -- 操作记录表 CREATE TABLE Actions ( ActionID INT PRIMARY KEY AUTO_INCREMENT, UserID INT, ActionType VARCHAR(100), Timestamp DATETIME DEFAULT CURRENT_TIMESTAMP, Status VARCHAR(20), FOREIGN KEY (UserID) REFERENCES Users(UserID) ); -- 审计日志表 CREATE TABLE AuditLogs ( LogID INT PRIMARY KEY AUTO_INCREMENT, ActionID INT, ReviewerID INT, Notes TEXT, FOREIGN KEY (ActionID) REFERENCES Actions(ActionID), FOREIGN KEY (ReviewerID) REFERENCES Users(UserID) ); -- 通知表 CREATE TABLE Notifications ( NotificationID INT PRIMARY KEY AUTO_INCREMENT, UserID INT, Message TEXT, SentDate DATETIME DEFAULT CURRENT_TIMESTAMP, Status VARCHAR(20) DEFAULT 'Unread', FOREIGN KEY (UserID) REFERENCES Users(UserID) ); -- 反馈表 CREATE TABLE Feedback ( FeedbackID INT PRIMARY KEY AUTO_INCREMENT, UserID INT, Content TEXT, SubmittedDate DATETIME DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (UserID) REFERENCES Users(UserID) ); -- 政策更新表 CREATE TABLE PolicyUpdates ( UpdateID INT PRIMARY KEY AUTO_INCREMENT, PolicyID INT, UpdatedBy INT, UpdateDate DATETIME DEFAULT CURRENT_TIMESTAMP, ChangeDescription TEXT, FOREIGN KEY (PolicyID) REFERENCES Policies(PolicyID), FOREIGN KEY (UpdatedBy) REFERENCES Users(UserID) ); --------------------------------------------------- -- 3. 创建整合视图 --------------------------------------------------- CREATE VIEW UserSummary AS SELECT u.UserID, u.Name AS 用户姓名, u.Email AS 邮箱, u.Department AS 部门, r.RoleName AS 角色, a.ActionType AS 操作类型, a.Timestamp AS 操作时间, a.Status AS 操作状态, n.Message AS 通知内容, n.SentDate AS 通知时间, n.Status AS 通知状态, f.Content AS 反馈内容, f.SubmittedDate AS 反馈时间 FROM Users u LEFT JOIN Roles r ON u.RoleID = r.RoleID LEFT JOIN Actions a ON u.UserID = a.UserID LEFT JOIN Notifications n ON u.UserID = n.UserID LEFT JOIN Feedback f ON u.UserID = f.UserID;
