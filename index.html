SELECT u.UserID, u.Name AS 用户姓名, u.Email AS 邮箱, u.Department AS 部门, a.ActionType AS 操作类型, a.Timestamp AS 操作时间, a.Status AS 操作状态, n.Message AS 通知内容, n.SentDate AS 通知时间, n.Status AS 通知状态, f.Content AS 反馈内容, f.SubmittedDate AS 反馈时间 FROM Users u LEFT JOIN Actions a ON u.UserID = a.UserID LEFT JOIN Notifications n ON u.UserID = n.UserID LEFT JOIN Feedback f ON u.UserID = f.UserID WHERE u.Name = '张三';
